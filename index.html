// Example: Modern Razorpay/UPI Intent Trigger
function openUpiGateway(amount, orderId) {
  const options = {
    "key": "YOUR_RAZORPAY_KEY", // From Payment Dashboard
    "amount": amount * 100, // Amount in paise
    "currency": "INR",
    "name": "My Offline Business",
    "description": "Order Payment #" + orderId,
    "upi": {
      "flow": "intent" // Automatically opens GPay/PhonePe on mobile
    },
    "handler": function (response){
      alert("Payment Successful ID: " + response.razorpay_payment_id);
    },
    "prefill": {
      "contact": "9999999999"
    }
  };
  const rzp = new Razorpay(options);
  rzp.open();
}
